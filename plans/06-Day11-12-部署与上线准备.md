# CORS 配置 — 从环境变量读取允许的来源
allowed_origins_str = os.getenv("ALLOWED_ORIGINS", "http://localhost:3000")

### 任务 9.12：添加安全头（后端）

创建 `backend/app/middleware/security.py`：

```python
from fastapi import Request
from fastapi.responses import Response


async def security_headers_middleware(request: Request, call_next):
    """添加安全相关的 HTTP 头"""
    response: Response = await call_next(request)
    
    # 防止点击劫持
    response.headers["X-Frame-Options"] = "DENY"
    # 防止 MIME 类型嗅探
    response.headers["X-Content-Type-Options"] = "nosniff"
    # 启用 XSS 过滤器
    response.headers["X-XSS-Protection"] = "1; mode=block"
    # 内容安全策略（宽松版，V1 阶段）
    response.headers["Content-Security-Policy"] = "default-src 'self'"
    # 引用策略
    response.headers["Referrer-Policy"] = "strict-origin-when-cross-origin"
    
    return response
```

在 `backend/app/main.py` 中注册：

```python
from app.middleware.security import security_headers_middleware

# 在 CORS 之后添加
app.middleware("http")(security_headers_middleware)
```

### 任务 9.13：添加速率限制中间件注册

确保 `backend/app/middleware/rate_limit.py` 中的中间件已在 `main.py` 中注册：

```python
from app.middleware.rate_limit import rate_limit_middleware

# 在 security headers 之后添加
app.middleware("http")(rate_limit_middleware)
```

**中间件注册顺序（重要）：**
```
CORS → Security Headers → Rate Limit → 业务路由
```

### 任务 9.14：生产环境验证脚本

创建 `backend/tests/test_production.py`：

```python
"""
生产环境验证脚本
使用方法：PROD_URL=https://betweenlines-api.railway.app python tests/test_production.py
"""
import os
import httpx
import asyncio

PROD_URL = os.getenv("PROD_URL", "http://localhost:8000")

SAMPLE_CHAT = """A: 今天在干嘛呀
B: 刚下班，好累哈哈
A: 辛苦啦，吃饭了吗
B: 还没呢"""


async def test_health():
    async with httpx.AsyncClient(timeout=10.0) as client:
        response = await client.get(f"{PROD_URL}/health")
        print(f"✅ Health: {response.status_code} - {response.json()}")
        assert response.status_code == 200


async def test_cors():
    async with httpx.AsyncClient(timeout=10.0) as client:
        response = await client.options(
            f"{PROD_URL}/api/v1/analyze",
            headers={
                "Origin": "https://betweenlines.vercel.app",
                "Access-Control-Request-Method": "POST",
            },
        )
        print(f"✅ CORS preflight: {response.status_code}")
        # 检查 CORS 头
        assert "access-control-allow-origin" in response.headers


async def test_analyze():
    async with httpx.AsyncClient(timeout=60.0) as client:
        response = await client.post(
            f"{PROD_URL}/api/v1/analyze",
            json={"chat_content": SAMPLE_CHAT},
        )
        data = response.json()
        print(f"✅ Analyze: {response.status_code} - status={data.get('chat_status')}")
        assert response.status_code == 200
        assert "chat_status" in data
        assert "reply_suggestions" in data


async def test_rate_limit():
    async with httpx.AsyncClient(timeout=10.0) as client:
        responses = []
        for i in range(25):  # 超过每分钟 20 次限制
            response = await client.post(
                f"{PROD_URL}/api/v1/analyze",
                json={"chat_content": "A: hi\nB: hello"},
            )
            responses.append(response.status_code)
        
        has_429 = 429 in responses
        print(f"✅ Rate limit: {'429 returned' if has_429 else 'No rate limiting detected'}")
        # V1 阶段不强制要求，但建议有


async def main():
    print("=" * 50)
    print(f"Production Validation: {PROD_URL}")
    print("=" * 50)
    
    await test_health()
    await test_cors()
    await test_analyze()
    await test_rate_limit()
    
    print("\n✅ All production tests completed!")


if __name__ == "__main__":
    asyncio.run(main())
```
